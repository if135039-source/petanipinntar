```javascript // ========================================== // AGRI-SIAGA // Deteksi lokasi pengguna otomatis // ========================================== let latitude = null; let longitude = null; // ========================================== // Memulai aplikasi // ========================================== document.addEventListener("DOMContentLoaded", () => { const tombolLokasi = document.getElementById("btnLokasi"); if (tombolLokasi) { tombolLokasi.addEventListener( "click", ambilLokasi ); } // Coba meminta lokasi otomatis ambilLokasi(); }); // ========================================== // Meminta izin lokasi pengguna // ========================================== function ambilLokasi() { const status = document.getElementById("statusLokasi"); if (!navigator.geolocation) { if (status) { status.textContent = "❌ Browser tidak mendukung lokasi."; } return; } if (status) { status.textContent = "📍 Meminta izin lokasi..."; } navigator.geolocation.getCurrentPosition( // Jika berhasil posisiBerhasil, // Jika gagal posisiGagal, { enableHighAccuracy: true, timeout: 10000, maximumAge: 300000 } ); } // ========================================== // Lokasi berhasil didapatkan // ========================================== function posisiBerhasil(position) { latitude = position.coords.latitude; longitude = position.coords.longitude; console.log("Latitude:", latitude); console.log("Longitude:", longitude); const status = document.getElementById("statusLokasi"); if (status) { status.textContent = "✅ Lokasi berhasil ditemukan"; } // Tampilkan koordinat const koordinat = document.getElementById("koordinat"); if (koordinat) { koordinat.textContent = `📍 ${latitude.toFixed(5)}, ${longitude.toFixed(5)}`; } // Cari nama daerah cariNamaLokasi(latitude, longitude); // Ambil data cuaca ambilCuaca(latitude, longitude); } // ========================================== // Lokasi gagal // ========================================== function posisiGagal(error) { const status = document.getElementById("statusLokasi"); let pesan = "❌ Lokasi tidak dapat ditemukan."; switch (error.code) { case error.PERMISSION_DENIED: pesan = "❌ Izin lokasi ditolak. Silakan aktifkan lokasi di browser."; break; case error.POSITION_UNAVAILABLE: pesan = "❌ Informasi lokasi tidak tersedia."; break; case error.TIMEOUT: pesan = "❌ Waktu pencarian lokasi habis."; break; } if (status) { status.textContent = pesan; } console.error( "Geolocation error:", error.message ); } // ========================================== // Mencari nama daerah dari koordinat // ========================================== async function cariNamaLokasi(lat, lon) { try { const url = `https://nominatim.openstreetmap.org/reverse?format=json&lat=${lat}&lon=${lon}&zoom=10&addressdetails=1`; const response = await fetch(url); if (!response.ok) { throw new Error( "Gagal mendapatkan nama lokasi" ); } const data = await response.json(); const address = data.address || {}; const kabupaten = address.county || address.city || address.town || address.municipality || "Tidak diketahui"; const provinsi = address.state || "Tidak diketahui"; tampilkanLokasi( kabupaten, provinsi ); } catch (error) { console.error( "Reverse geocoding error:", error ); const namaWilayah = document.getElementById("namaWilayah"); if (namaWilayah) { namaWilayah.textContent = "Lokasi ditemukan"; } } } // ========================================== // Menampilkan nama lokasi // ========================================== function tampilkanLokasi( wilayah, provinsi ) { const nama = document.getElementById("namaWilayah"); const prov = document.getElementById("provinsi"); if (nama) { nama.textContent = wilayah; } if (prov) { prov.textContent = provinsi; } } // ========================================== // Mengambil data cuaca // ========================================== // // Menggunakan Open-Meteo. // Tidak membutuhkan API key untuk penggunaan // non-komersial tertentu. // ========================================== async function ambilCuaca(lat, lon) { const statusCuaca = document.getElementById("statusCuaca"); if (statusCuaca) { statusCuaca.textContent = "🌦️ Mengambil data cuaca..."; } try { const url = `https://api.open-meteo.com/v1/forecast?latitude=${lat}&longitude=${lon}¤t=temperature_2m,relative_humidity_2m,precipitation,rain,weather_code&daily=precipitation_sum&timezone=auto`; const response = await fetch(url); if (!response.ok) { throw new Error( "Gagal mengambil data cuaca" ); } const data = await response.json(); const current = data.current; const daily = data.daily; // Suhu const suhu = current.temperature_2m; // Kelembapan const kelembapan = current.relative_humidity_2m; // Hujan saat ini const hujan = current.precipitation; // Total prediksi hujan beberapa hari const curahHujan = daily.precipitation_sum; tampilkanCuaca( suhu, kelembapan, hujan, curahHujan ); analisisMusim( curahHujan, suhu, kelembapan ); if (statusCuaca) { statusCuaca.textContent = "✅ Data cuaca berhasil diperbarui"; } } catch (error) { console.error( "Weather API error:", error ); if (statusCuaca) { statusCuaca.textContent = "❌ Data cuaca gagal diambil."; } } } // ========================================== // Menampilkan data cuaca // ========================================== function tampilkanCuaca( suhu, kelembapan, hujan, curahHujan ) { const suhuElement = document.getElementById("suhu"); const kelembapanElement = document.getElementById("kelembapan"); const hujanElement = document.getElementById("hujan"); const curahElement = document.getElementById("curahHujan"); if (suhuElement) { suhuElement.textContent = `${suhu} °C`; } if (kelembapanElement) { kelembapanElement.textContent = `${kelembapan}%`; } if (hujanElement) { hujanElement.textContent = `${hujan} mm`; } if (curahElement) { const total = curahHujan.reduce( (a, b) => a + b, 0 ); curahElement.textContent = `${total.toFixed(1)} mm`; } } // ========================================== // Analisis kondisi // ========================================== function analisisMusim( curahHujan, suhu, kelembapan ) { const totalHujan = curahHujan.reduce( (a, b) => a + b, 0 ); let kondisi; let icon; let warna; let rekomendasi; // Kondisi basah if (totalHujan >= 100) { kondisi = "Cenderung Basah"; icon = "🌧️"; warna = "#2196f3"; rekomendasi = "Pantau genangan air, pastikan drainase berjalan baik, dan perhatikan risiko penyakit tanaman akibat kelembapan tinggi."; } // Kondisi peralihan else if (totalHujan >= 40) { kondisi = "Peralihan"; icon = "🌤️"; warna = "#ff9800"; rekomendasi = "Kondisi cuaca sedang berubah. Pantau hujan dan sesuaikan penyiraman dengan kebutuhan tanaman."; } // Kondisi kering else { kondisi = "Cenderung Kering"; icon = "☀️"; warna = "#f44336"; rekomendasi = "Curah hujan relatif rendah. Perhatikan ketersediaan air dan lakukan penyiraman secara efisien."; } tampilkanAnalisis( kondisi, icon, warna, rekomendasi ); } // ========================================== // Menampilkan hasil analisis // ========================================== function tampilkanAnalisis( kondisi, icon, warna, rekomendasi ) { const kondisiElement = document.getElementById("kondisi"); const iconElement = document.getElementById("iconMusim"); const rekomendasiElement = document.getElementById("rekomendasi"); if (kondisiElement) { kondisiElement.textContent = kondisi; kondisiElement.style.color = warna; } if (iconElement) { iconElement.textContent = icon; } if (rekomendasiElement) { rekomendasiElement.textContent = rekomendasi; } } // ========================================== // Tombol refresh lokasi // ========================================== function refreshLokasi() { const status = document.getElementById("statusLokasi"); if (status) { status.textContent = "🔄 Memperbarui lokasi..."; } ambilLokasi(); } // Agar bisa dipanggil dari HTML window.refreshLokasi = refreshLokasi; ```
